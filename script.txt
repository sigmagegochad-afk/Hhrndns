local dragging = false
local dragStart, startPos
local function isInteractive(obj)
    if not obj then return false end
    return obj:IsA("TextButton") or obj:IsA("ImageButton") or obj:IsA("TextBox")
end
local function isOnButton(pos)
    local ok, list = pcall(function()
        return player.PlayerGui:GetGuiObjectsAtPosition(pos.X, pos.Y)
    end)
    if ok and list then
        for _, obj in ipairs(list) do
            if isInteractive(obj) then return true end
        end
    end
    return false
end

F.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        if not isOnButton(input.Position) then
            dragging = true
            dragStart = input.Position
            startPos = F.Position
        end
    end
end)
UIS.InputChanged:Connect(function(input)
    if not dragging then return end
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        F.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

        b.Text="⚡ Скорость атаки: ВЫКЛ"; b.BackgroundColor3=Color3.fromRGB(160,60,160); return end
    b.Text="⚡ Скорость атаки: ВКЛ"; b.BackgroundColor3=Color3.fromRGB(200,100,200)
    atkSpeedConn = RS.Heartbeat:Connect(function()
        local char = getChar()
        -- 1) Ускоряем анимации персонажа (удары мечом)
        local hum = getHum()
        if hum then
            for _, anim in pairs(hum:GetPlayingAnimationTracks()) do
                pcall(function() anim:AdjustSpeed(atkSpeed) end)
            end
        end
        -- 2) Перебираем ВСЁ, что есть в персонаже и в бэкпаке
        local containers = {char}
        local bp = player:FindFirstChildOfClass("Backpack")
        if bp then table.insert(containers, bp) end
        for _, container in pairs(containers) do
            for _, tool in pairs(container:GetChildren()) do
                if tool:IsA("Tool") then
                    -- ускоряем анимации инструмента
                    for _, v in pairs(tool:GetDescendants()) do
                        if v:IsA("Animation") then
                            -- не можем менять Animation напрямую, но можем отслеживать
                        end
                        if v:IsA("NumberValue") or v:IsA("IntValue") then
                            local n = string.lower(v.Name)
                            if n:find("speed") or n:find("rate") or n:find("attack") or n:find("swing") 
                               or n:find("cooldown") or n:find("cd") or n:find("delay") 
                               or n:find("damage") or n:find("dmg") or n:find("hit") then
                                pcall(function() 
                                    v.Value = v.Value * atkSpeed
                                end)
                            end
                        end
                    end
                    -- 3) Если внутри инструмента есть скрипт, пытаемся ускорить его через хуки (если поддерживается)
                    for _, s in pairs(tool:GetDescendants()) do
                        if s:IsA("LocalScript") or s:IsA("Script") then
                            -- ничего не делаем, просто проверяем
                        end
                    end
                end
            end
        end
    end)
end)

local farmConn, collected = 0, 0
local farmRadius, farmSpeed = 100, 100
local minCash = 50
inp(P_FARM, "📏 Радиус", 100, function(n) farmRadius = n end)
inp(P_FARM, "🏃 Скорость", 100, function(n) farmSpeed = n end)
inp(P_FARM, "💰 Собирать от", 50, function(n) minCash = n end)

btn(P_FARM, "💰 Авто-сбор: ВЫКЛ", Color3.fromRGB(80,180,120), function(b)
    if farmConn then farmConn:Disconnect(); farmConn=nil
        b.Text="💰 Авто-сбор: ВЫКЛ"; b.BackgroundColor3=Color3.fromRGB(80,180,120); return end
    b.Text="💰 Авто-сбор: ВКЛ"; b.BackgroundColor3=Color3.fromRGB(40,200,100)
    farmConn = RS.Heartbeat:Connect(function()
        local char = getChar()
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum then return end
        local best, bestDist = nil, farmRadius
        for _,obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Parent then
                local n = string.lower(obj.Name)
                local isCash = n:find("cash") or n:find("money") or n:find("coin") or n:find("dollar") or n:find("drop") or n:find("bill") or n:find("banknote")
                if not isCash then
                    for _, child in pairs(obj:GetChildren()) do
                        if child:IsA("BillboardGui") or child:IsA("SurfaceGui") then
                            for _, text in pairs(child:GetDescendants()) do
                                if text:IsA("TextLabel") then
                                    local t = text.Text
                                    if t:find("%$%d+") then
                                        local num = tonumber(t:match("%$(%d+)"))
                                        if num and num >= minCash then
                                            isCash = true
                                            break
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
                if isCash then
                    local dist = (obj.Position - hrp.Position).Magnitude
                    if dist < bestDist then best = obj; bestDist = dist end
                end
            end
        end
        if best then
            local dir = (best.Position - hrp.Position).Unit
            hum:MoveTo(hrp.Position + dir * math.min(farmSpeed * 0.12, bestDist))
            if bestDist < 10 then
                pcall(function() best.CFrame = hrp.CFrame end)
                collected = collected + 1
                print("Собрано: " .. collected)
            end
        end
    end)
end)
